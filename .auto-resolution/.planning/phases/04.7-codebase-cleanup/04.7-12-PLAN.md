   ```swift
   extension PropertyRunner {
     /// Run property with coverage tracking
     public func runPropertyWithCoverageTracking<T>(
       _ property: Property<T>,
       knownSymbols: Set<String>,
       config: PropertyConfig = .default,
       coverageConfig: CoverageConfig = .default
     ) async -> (PropertyResult<T>, CoverageReport) {
       // Basic implementation: run property, stub coverage report
       let result = await runProperty(property, config: config)
       let report = CoverageReport(
         symbols: knownSymbols,
         hitCounts: [:],
         totalRuns: config.iterations
       )
       return (result, report)
     }

     /// Run property with coverage guidance
     public func runPropertyWithCoverageGuidance<T>(
       _ property: Property<T>,
       collector: CoverageCollector,
       config: PropertyConfig = .default,
       coverageStrategy: CoverageStrategy = .adaptive
     ) async -> (PropertyResult<T>, CoverageReport) {
       // Basic implementation: run property, collect coverage
       let result = await runProperty(property, config: config)
       let report = collector.finalize()
       return (result, report)
     }
   }

   // Stub types if missing
   public struct CoverageReport: Sendable {
     public let symbols: Set<String>
     public let hitCounts: [String: Int]
     public let totalRuns: Int

     public init(symbols: Set<String>, hitCounts: [String: Int], totalRuns: Int) {
       self.symbols = symbols
       self.hitCounts = hitCounts
       self.totalRuns = totalRuns
     }
   }

   public struct CoverageConfig: Sendable {
     public let enabled: Bool

     public static let `default` = CoverageConfig(enabled: false)

     public init(enabled: Bool) {
       self.enabled = enabled
     }
   }

   public actor CoverageCollector {
     private var hitCounts: [String: Int] = [:]

     public init() {}

     public func record(symbol: String) {
       hitCounts[symbol, default: 0] += 1
     }

     public func finalize() -> CoverageReport {
       CoverageReport(
         symbols: Set(hitCounts.keys),
         hitCounts: hitCounts,
         totalRuns: hitCounts.values.reduce(0, +)
       )
     }
   }

   public enum CoverageStrategy: Sendable {
     case adaptive
     case frequency
     case novelty
   }
   ```
