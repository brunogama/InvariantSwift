   ```swift
   extension ShrinkTree {
     /// Breadth-first traversal of shrink tree
     public func breadthFirst() -> [T] {
       var result: [T] = [root]
       var queue: [ShrinkTree<T>] = [self]
       var index = 0

       while index < queue.count {
         let node = queue[index]
         for child in node.children {
           result.append(child.root)
           queue.append(child)
         }
         index += 1
       }

       return result
     }

     /// Depth-first traversal of shrink tree
     public func depthFirst() -> [T] {
       var result: [T] = [root]
       for child in children {
         result.append(contentsOf: child.depthFirst())
       }
       return result
     }
   }
   ```

   ```swift
   public func expectNoDifference(_ actual: String, _ expected: String, file: StaticString = #file, line: UInt = #line) {
     guard actual != expected else { return }

     let actualLines = actual.split(separator: "\n", omittingEmptySubsequences: false)
     let expectedLines = expected.split(separator: "\n", omittingEmptySubsequences: false)

     var diff = "Strings differ:\n"
     let maxLines = max(actualLines.count, expectedLines.count)

     for i in 0..<maxLines {
       let actualLine = i < actualLines.count ? String(actualLines[i]) : ""
       let expectedLine = i < expectedLines.count ? String(expectedLines[i]) : ""

       if actualLine != expectedLine {
         diff += "\nLine \(i + 1):\n"
         diff += "  Expected: \(expectedLine)\n"
         diff += "  Actual:   \(actualLine)\n"
       }
     }

     preconditionFailure(diff, file: file, line: line)
   }
   ```
