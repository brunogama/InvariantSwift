Recipe 6: Using Coverage Requirements
```swift
@PropertyTest
func testSortingWithCoverage(array: [Int]) {
    Property(generator: Gen.array(Gen<Int>.int)) { arr in
        arr.sorted().count == arr.count
    }
    .cover(30, when: { $0.isEmpty }, label: "empty arrays")
    .cover(50, when: { $0.count > 5 }, label: "non-trivial arrays")
}
```

Recipe 7: Classifying Test Inputs
```swift
@PropertyTest
func testWithClassification(number: Int) {
    Property(generator: Gen<Int>.int(in: -100...100)) { n in
        abs(n) >= 0
    }
    .classify(when: { $0 > 0 }, label: "positive")
    .classify(when: { $0 < 0 }, label: "negative")
    .classify(when: { $0 == 0 }, label: "zero")
}
```

Recipe 8: Labeling for Debug Output
```swift
@PropertyTest
func testWithLabels(user: User) {
    Property(generator: User.arbitrary) { user in
        user.validate().isValid
    }
    .label { "Testing user: \($0.name)" }
}
```

Recipe 9: Collecting Value Distributions
```swift
@PropertyTest
func testArrayLengthDistribution(array: [String]) {
    Property(generator: Gen.array(Gen<String>.string)) { arr in
        !arr.contains("")  // Some property
    }
    .collect { $0.count }  // Histogram of array lengths
}
```

Recipe 10: Conditional Properties with ==>
```swift
@PropertyTest
func testDivision(a: Int, b: Int) {
    // Only test when b != 0
    (b != 0) ==> {
        let result = a / b
        return result * b + (a % b) == a
    }
}
```

**New Section: Real-World Application Recipes**

Recipe 11: API Response Validation
```swift
@PropertyTest
func testAPIResponseParsing(response: APIResponse) {
    Property(generator: APIResponse.arbitrary) { response in
        let encoded = try! JSONEncoder().encode(response)
        let decoded = try! JSONDecoder().decode(APIResponse.self, from: encoded)
        return decoded == response
    }
    .cover(20, when: { $0.error != nil }, label: "error responses")
    .cover(50, when: { !$0.data.isEmpty }, label: "with data")
}
```

Recipe 12: State Machine Testing
```swift
@PropertyTest
func testOrderStateMachine(events: [OrderEvent]) {
    Property(generator: Gen.array(OrderEvent.arbitrary)) { events in
        var order = Order()
        for event in events {
            order.apply(event)
        }
        return order.isConsistent
    }
    .classify(when: { $0.contains(.cancel) }, label: "with cancellation")
    .classify(when: { $0.contains(.refund) }, label: "with refund")
}
```

Recipe 13: Database Roundtrip
```swift
@PropertyTest
func testDatabaseRoundtrip(entity: Entity) {
    Property(generator: Entity.arbitrary) { entity in
        let id = database.insert(entity)
        let retrieved = database.fetch(id)
        return retrieved == entity
    }
    .cover(30, when: { $0.optionalField == nil }, label: "null optionals")
}
```

Recipe 14: Validation Rules
```swift
@PropertyTest
func testEmailValidation(email: String) {
    Property(generator: Gen.email) { email in
        let result = EmailValidator.validate(email)
        return result.isValid == email.contains("@")
    }
    .classify(when: { $0.contains("+") }, label: "plus addressing")
    .classify(when: { $0.hasSuffix(".com") }, label: ".com domain")
}
```

Recipe 15: Concurrent Data Structure
```swift
@PropertyTest
func testConcurrentQueue(operations: [QueueOp]) async {
    Property(generator: Gen.array(QueueOp.arbitrary)) { ops in
        let queue = ConcurrentQueue<Int>()
        await withTaskGroup(of: Void.self) { group in
            for op in ops {
                group.addTask { await queue.execute(op) }
            }
        }
        return queue.isConsistent
    }
    .cover(30, when: { $0.filter { $0.isDequeue }.count > 5 }, label: "heavy dequeue")
}
```
