/// Configures shrink behavior for a property test parameter.
///
/// Use this attribute on parameters to guide shrinking toward
/// semantically meaningful values.
///
/// - Parameter target: The value to shrink toward
///
/// Usage:
/// ```swift
/// @PropertyTest
/// func testPositiveCount(@ShrinkTowards(1) count: Int) -> Bool {
///   // count will shrink toward 1, not 0
///   count > 0
/// }
///
/// @PropertyTest
/// func testNonEmptyName(@ShrinkTowards("test") name: String) -> Bool {
///   // name will shrink toward "test", not ""
///   !name.isEmpty
/// }
/// ```
@attached(accessor)
public macro ShrinkTowards<T>(_ target: T) = #externalMacro(
  module: "InvariantSwiftMacros",
  type: "ShrinkTowardsMacro"
)

// Convenience for zero
@attached(accessor)
public macro ShrinkTowards(_ target: ShrinkTarget<some Any>) = #externalMacro(
