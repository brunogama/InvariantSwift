```swift
extension SVGChartGenerator {
  /// Generates a histogram SVG from label distribution data.
  public func distributionHistogram(
    labels: [(String, Int)],
    title: String = "Input Distribution"
  ) -> String {
    let maxCount = labels.map { $0.1 }.max() ?? 1
    let barWidth = (config.width - 100) / max(1, labels.count)
    let scaleY = Double(config.height - 60) / Double(maxCount)

    var svg = """
    <svg width="\(config.width)" height="\(config.height)"
         xmlns="http://www.w3.org/2000/svg">
      <rect width="100%" height="100%" fill="\(config.backgroundColor)"/>
      <text x="\(config.width/2)" y="20" text-anchor="middle"
            font-size="\(config.fontSize + 2)" fill="\(config.labelColor)">\(title)</text>
    """

    for (index, (label, count)) in labels.enumerated() {
      let x = 50 + index * barWidth
      let barHeight = Int(Double(count) * scaleY)
      let y = config.height - 40 - barHeight

      svg += """
        <rect x="\(x)" y="\(y)" width="\(barWidth - 4)"
              height="\(barHeight)" fill="\(config.barColor)"/>
        <text x="\(x + barWidth/2)" y="\(config.height - 25)"
              text-anchor="middle" font-size="\(config.fontSize)"
              fill="\(config.labelColor)">\(escapeXML(label))</text>
        <text x="\(x + barWidth/2)" y="\(y - 5)"
              text-anchor="middle" font-size="\(config.fontSize)"
              fill="\(config.labelColor)">\(count)</text>
      """
    }

    svg += "</svg>"
    return svg
  }

  private func escapeXML(_ string: String) -> String {
    string
      .replacingOccurrences(of: "&", with: "&amp;")
      .replacingOccurrences(of: "<", with: "&lt;")
      .replacingOccurrences(of: ">", with: "&gt;")
      .replacingOccurrences(of: "\"", with: "&quot;")
  }
}
```

```swift
extension SVGChartGenerator {
  /// Generates a shrinking path visualization showing how value was reduced.
  public func shrinkingPath<T: CustomStringConvertible>(
    steps: [T],
    title: String = "Shrinking Path"
  ) -> String {
    let nodeRadius = 20
    let spacing = (config.width - 100) / max(1, steps.count)

    var svg = """
    <svg width="\(config.width)" height="\(config.height)"
         xmlns="http://www.w3.org/2000/svg">
      <rect width="100%" height="100%" fill="\(config.backgroundColor)"/>
      <text x="\(config.width/2)" y="20" text-anchor="middle"
            font-size="\(config.fontSize + 2)">\(title)</text>
    """

    let centerY = config.height / 2

    for (index, value) in steps.enumerated() {
      let x = 50 + index * spacing

      // Draw connecting line
      if index > 0 {
        let prevX = 50 + (index - 1) * spacing
        svg += """
          <line x1="\(prevX + nodeRadius)" y1="\(centerY)"
                x2="\(x - nodeRadius)" y2="\(centerY)"
                stroke="#999" stroke-width="2"/>
          <polygon points="\(x - nodeRadius - 5),\(centerY - 5) \(x - nodeRadius),\(centerY) \(x - nodeRadius - 5),\(centerY + 5)"
                   fill="#999"/>
        """
      }

      // Draw node
      let isFirst = index == 0
      let isLast = index == steps.count - 1
      let fillColor = isFirst ? "#ff6b6b" : (isLast ? "#4CAF50" : "#fff")

      svg += """
        <circle cx="\(x)" cy="\(centerY)" r="\(nodeRadius)"
                fill="\(fillColor)" stroke="#333" stroke-width="2"/>
        <text x="\(x)" y="\(centerY + nodeRadius + 15)"
              text-anchor="middle" font-size="\(config.fontSize)">\(escapeXML(String(describing: value)))</text>
      """
    }

    svg += "</svg>"
    return svg
  }
}
```

```swift
extension HTMLReportGenerator {
  /// Generates a complete HTML report.
  public func generate(from data: ReportData) -> String {
    """
    <!DOCTYPE html>
    <html lang="en">
    <head>
      <meta charset="UTF-8">
      <meta name="viewport" content="width=device-width, initial-scale=1.0">
      <title>Property Test Report: \(escapeHTML(data.testName))</title>
      <style>
        \(embeddedCSS)
      </style>
    </head>
    <body>
      <div class="container">
        \(headerSection(data))
        \(summarySection(data))
        \(distributionSection(data))
        \(shrinkingSection(data))
        \(detailsSection(data))
        \(footerSection(data))
      </div>
    </body>
    </html>
    """
  }

  private var embeddedCSS: String {
    """
    * { margin: 0; padding: 0; box-sizing: border-box; }
    body { font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif;
           line-height: 1.6; color: #333; background: #f5f5f5; }
    .container { max-width: 900px; margin: 0 auto; padding: 20px; }
    .header { background: #2c3e50; color: white; padding: 20px; border-radius: 8px 8px 0 0; }
    .section { background: white; padding: 20px; margin-bottom: 1px; }
    .status-passed { color: #27ae60; }
    .status-failed { color: #e74c3c; }
    .status-gave_up { color: #f39c12; }
    .metric { display: inline-block; margin-right: 30px; }
    .metric-value { font-size: 24px; font-weight: bold; }
    .metric-label { font-size: 12px; color: #666; }
    pre { background: #f8f9fa; padding: 15px; border-radius: 4px; overflow-x: auto; }
    .chart { margin: 20px 0; }
    """
  }

  private func headerSection(_ data: ReportData) -> String {
    let statusClass = "status-\(data.result)"
    return """
    <div class="header">
      <h1>\(escapeHTML(data.testName))</h1>
      <p class="\(statusClass)">\(data.result.uppercased())</p>
    </div>
    """
  }

  private func summarySection(_ data: ReportData) -> String {
    // Generate metrics summary
  }

  private func distributionSection(_ data: ReportData) -> String {
    guard let classification = data.classification,
          !classification.labels.isEmpty else { return "" }

    let labels = classification.labels.map { ($0.key, $0.value) }
    let chart = chartGenerator.distributionHistogram(labels: labels)

    return """
    <div class="section">
      <h2>Input Distribution</h2>
      <div class="chart">\(chart)</div>
    </div>
    """
  }

    guard let path = data.shrinkPath, !path.isEmpty else { return "" }

    return """
    <div class="section">
      <h2>Shrinking Path</h2>
      <div class="chart">\(chart)</div>
    </div>
    """

      .replacingOccurrences(of: "<", with: "&lt;")
      .replacingOccurrences(of: ">", with: "&gt;")
  }
}
```

```swift
extension PropertyResult {
  /// Exports test result as HTML report to a file.
  ///
  /// - Parameters:
  ///   - url: File URL to write the report
  ///   - testName: Name of the test for the report header
  ///   - classification: Optional classification report for distribution chart
  ///   - shrinkPath: Optional shrink path for visualization
  ///
  /// - Throws: File write errors
  public func exportHTML(
    to url: URL,
    testName: String,
    classification: ClassificationReport? = nil,
    shrinkPath: [String]? = nil
  ) throws {
    let generator = HTMLReportGenerator()

    let data = HTMLReportGenerator.ReportData(
      testName: testName,
      result: resultString,
      iterations: iterationCount,
      shrinkSteps: shrinkStepCount,
      counterexample: counterexampleDescription,
      shrunkValue: shrunkDescription,
      classification: classification,
      shrinkPath: shrinkPath,
      timestamp: Date(),
      duration: nil,
      seed: seedValue
    )

    let html = generator.generate(from: data)
    try html.write(to: url, atomically: true, encoding: .utf8)
  }

  private var resultString: String {
    switch self {
    case .success: return "passed"
    case .failure: return "failed"
    case .gaveUp: return "gave_up"
    }
  }

  // ... other computed properties for data extraction
}
```
