public enum TimeoutDuration: Sendable, Equatable {
  case seconds(TimeInterval)
  case milliseconds(Int)
  case none  // Disables timeout (for debugging)

  public var nanoseconds: UInt64? {
    switch self {
    case .seconds(let s): return UInt64(s * 1_000_000_000)
    case .milliseconds(let ms): return UInt64(ms) * 1_000_000
    case .none: return nil
    }
  }
}

3. Implement withPropertyTimeout using the task racing pattern from RESEARCH.md:
```swift
public func withPropertyTimeout<T: Sendable>(
  _ duration: TimeoutDuration,
  operation: @escaping @Sendable () async throws -> T
) async throws -> T {
  guard let nanos = duration.nanoseconds else {
    // .none - no timeout, run directly
    return try await operation()
  }

  return try await withThrowingTaskGroup(of: T.self) { group in
    group.addTask { try await operation() }
    group.addTask {
      try await Task.sleep(nanoseconds: nanos)
      throw PropertyTimeoutError(elapsed: ..., limit: ...)
    }

    guard let result = try await group.next() else {
      throw CancellationError()
    }
    group.cancelAll()
    return result
  }
}
```

4. Add convenience overload for seconds:
```swift

  seconds: TimeInterval,
  operation: @escaping @Sendable () async throws -> T

```

/// Enforces a timeout on a property test.
///
/// Use this macro to prevent property tests from hanging. If the test
/// exceeds the specified timeout, it fails with a clear timeout message.
///
/// - Parameter seconds: Maximum execution time in seconds
/// - Parameter duration: TimeoutDuration for more control
///
/// Usage:
/// ```swift
/// @Timeout(seconds: 5.0)
/// @PropertyTest
/// func slowProperty(data: [Int]) -> Bool {
///   expensiveComputation(data)
/// }
///
/// @Timeout(.none) // Disable for debugging
/// @PropertyTest
/// func debugProperty(n: Int) -> Bool { ... }
/// ```
@attached(peer)
public macro Timeout(seconds: Double) = #externalMacro(
  module: "InvariantSwiftMacros",
  type: "TimeoutMacro"
)

@attached(peer)
public macro Timeout(_ duration: TimeoutDuration) = #externalMacro(
  module: "InvariantSwiftMacros",
  type: "TimeoutMacro"
)
```
