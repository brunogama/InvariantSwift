// MARK: - Compile Verifier
// Verifies generated Swift code compiles using swiftc -typecheck.

import Foundation

/// Errors from compile verification.
public enum CompileVerificationError: Error, CustomStringConvertible {
  case compilerNotFound
  case compilationFailed(errors: [CompileError])
  case timeout
  case processError(String)

  public var description: String {
    switch self {
    case .compilerNotFound:
      return "Swift compiler not found. Ensure Xcode or Swift toolchain is installed."
    case .compilationFailed(let errors):
      let errorMessages = errors.map { "  Line \($0.line): \($0.message)" }.joined(separator: "\n")
      return "Compilation failed:\n\(errorMessages)"
    case .timeout:
      return "Compilation timed out after 30 seconds."
    case .processError(let message):
      return "Process error: \(message)"
    }
  }
}

/// A single compilation error with location.
public struct CompileError: Sendable {
  public let line: Int
  public let column: Int
  public let message: String
  public let severity: Severity

  public enum Severity: String, Sendable {
    case error
    case warning
    case note
  }

  public init(line: Int, column: Int, message: String, severity: Severity = .error) {
    self.line = line
    self.column = column
    self.message = message
    self.severity = severity
  }
}

/// Result of compile verification.
public enum CompileVerificationResult: Sendable {
  case success
  case failed(errors: [CompileError])

  public var isSuccess: Bool {
    if case .success = self { return true }
    return false
  }
}

/// Verifies Swift code compiles using swiftc -typecheck.
public struct CompileVerifier: Sendable {

  /// Timeout for compilation in seconds.
  public static let defaultTimeout: TimeInterval = 30

  /// Verify that Swift code compiles.
  /// - Parameters:
  ///   - code: Swift source code to verify
  ///   - additionalImports: Extra imports to add (e.g., target module)
  ///   - timeout: Timeout in seconds
  /// - Returns: Verification result
  public static func verify(
    code: String,
    additionalImports: [String] = [],
    timeout: TimeInterval = defaultTimeout
  ) throws -> CompileVerificationResult {
    // Create temp file
    let tempDir = FileManager.default.temporaryDirectory
    let tempFile = tempDir.appendingPathComponent("GhostwriterVerify_\(UUID().uuidString).swift")

    // Build full source with imports
    var fullSource = ""
    for imp in additionalImports {
      fullSource += "import \(imp)\n"
    }
    fullSource += "\n"
    fullSource += code

    try fullSource.write(to: tempFile, atomically: true, encoding: .utf8)
    defer { try? FileManager.default.removeItem(at: tempFile) }

    // Run swiftc -typecheck
    let process = Process()
    process.executableURL = URL(fileURLWithPath: "/usr/bin/xcrun")
    process.arguments = ["swiftc", "-typecheck", tempFile.path]

    let errorPipe = Pipe()
    process.standardError = errorPipe
    process.standardOutput = FileHandle.nullDevice

    do {
      try process.run()
    } catch {
      throw CompileVerificationError.processError(error.localizedDescription)
    }

    // Wait with timeout
    let startTime = Date()
    while process.isRunning {
      if Date().timeIntervalSince(startTime) > timeout {
        process.terminate()
        throw CompileVerificationError.timeout
      }
      Thread.sleep(forTimeInterval: 0.1)
    }

    if process.terminationStatus == 0 {
      return .success
    }

    // Parse errors from stderr
    let errorData = errorPipe.fileHandleForReading.readDataToEndOfFile()
    let errorOutput = String(data: errorData, encoding: .utf8) ?? ""
    let errors = parseCompilerErrors(errorOutput)

    return .failed(errors: errors)
  }

  /// Parse compiler output into structured errors.
  private static func parseCompilerErrors(_ output: String) -> [CompileError] {
    var errors: [CompileError] = []

    // Pattern: filename:line:column: error: message
    let pattern = #"^.*?:(\d+):(\d+):\s*(error|warning|note):\s*(.*)$"#
    let regex = try? NSRegularExpression(pattern: pattern, options: .anchorsMatchLines)

    let range = NSRange(output.startIndex..<output.endIndex, in: output)
    regex?.enumerateMatches(in: output, options: [], range: range) { match, _, _ in
      guard let match = match else { return }

      let lineRange = Range(match.range(at: 1), in: output)
      let columnRange = Range(match.range(at: 2), in: output)
      let severityRange = Range(match.range(at: 3), in: output)
      let messageRange = Range(match.range(at: 4), in: output)

      guard let lineRange, let columnRange, let severityRange, let messageRange else { return }

      let line = Int(output[lineRange]) ?? 0
      let column = Int(output[columnRange]) ?? 0
      let severityStr = String(output[severityRange])
      let message = String(output[messageRange])

      let severity: CompileError.Severity
      switch severityStr {
      case "warning": severity = .warning
      case "note": severity = .note
      default: severity = .error
      }

      errors.append(CompileError(line: line, column: column, message: message, severity: severity))
    }

    // If parsing failed but we had non-zero exit, create generic error
    if errors.isEmpty && !output.isEmpty {
      errors.append(CompileError(line: 0, column: 0, message: output.prefix(500).description, severity: .error))
    }

    return errors
  }
}

for (sourceFile, types) in typesByFile {
  let testCode = generator.generateTestFile(types: types, sourceFile: sourceFile)
  let testsInFile = types.reduce(0) { $0 + generator.detectPatterns(for: $1).count }

  // Verify compilation unless skipped
  if !config.skipCompileTest && !config.dryRun {

      let verifyResult = try CompileVerifier.verify(

        additionalImports: ["Foundation"]

      switch verifyResult {
      case .success:

          print("    Compilation verified")

      case .failed(let errors):
        print("  Compilation verification failed for \(sourceFile):")
        for error in errors.prefix(5) {
          print("    Line \(error.line): \(error.message)")
        }
        if errors.count > 5 {
          print("    ... and \(errors.count - 5) more errors")
        }
        print("  Use --skip-compile-test to skip verification")
        continue  // Skip writing this file
      }
    } catch {
      print("  Verification error for \(sourceFile): \(error)")
      if !config.verbose {
        print("  Use --verbose for details or --skip-compile-test to skip")
      }
      continue  // Skip writing this file
    }
  }

  result.testsGenerated += testsInFile

  if config.dryRun {
    // ... existing dry run code ...
  } else {
    let outputFile = try writeTestFile(
      testCode,
      sourceFile: sourceFile,
      outputDirectory: config.outputDirectory
    )
    result.generatedFiles.append(outputFile)

    if config.verbose {
      print("    Wrote \(outputFile)")
    }
  }
}

// MARK: - Ghostwriter Compilation Tests
// Tests for FR-4.3: Compile-Test Infrastructure

import Foundation
import Testing
@testable import GhostwriterCLI

@Suite("Compile Verification Tests")
struct CompileVerificationTests {

  @Test("Valid Swift code verifies successfully")
  func validCodeCompiles() throws {
    let validCode = """
    import Foundation

    struct SimpleType {
      let value: Int
    }
    """

    let result = try CompileVerifier.verify(code: validCode)

    #expect(result.isSuccess)
  }

  @Test("Invalid Swift code fails verification")
  func invalidCodeFails() throws {
    let invalidCode = """
    struct BrokenType {
      let value: UnknownType  // This type doesn't exist
    }
    """

    let result = try CompileVerifier.verify(code: invalidCode)

    switch result {
    case .success:
      Issue.record("Expected compilation to fail")
    case .failed(let errors):
      #expect(!errors.isEmpty)
      // Should mention the unknown type
      let hasUnknownTypeError = errors.contains { $0.message.contains("UnknownType") || $0.message.contains("undeclared") || $0.message.contains("cannot find") }
      #expect(hasUnknownTypeError)
    }
  }

  @Test("Syntax errors are detected")
  func syntaxErrorsDetected() throws {
    let syntaxError = """
    struct Incomplete {
      let missing  // Missing type annotation
    }
    """

    let result = try CompileVerifier.verify(code: syntaxError)

    #expect(!result.isSuccess)
  }

  @Test("Error parsing extracts line numbers")
  func errorLineNumbers() throws {
    let codeWithError = """
    struct First {
      let ok: Int
    }

    struct Second {
      let bad: NonExistent
    }
    """

    let result = try CompileVerifier.verify(code: codeWithError)

    switch result {
    case .success:
      Issue.record("Expected compilation to fail")
    case .failed(let errors):
      // Error should be around line 6 (where NonExistent is)
      let hasLineInfo = errors.contains { $0.line > 0 }
      #expect(hasLineInfo)
    }
  }

  @Test("Additional imports are included")
  func additionalImports() throws {
    let codeNeedingFoundation = """
    struct WithData {
      let data: Data
    }
    """

    // Without Foundation import, Data is unknown
    let withoutImport = try CompileVerifier.verify(code: codeNeedingFoundation)

    // With Foundation import, should compile
    let withImport = try CompileVerifier.verify(
      code: codeNeedingFoundation,
      additionalImports: ["Foundation"]
    )

    #expect(!withoutImport.isSuccess)
    #expect(withImport.isSuccess)
  }

  @Test("CompileVerificationResult isSuccess helper")
  func isSuccessHelper() {
    let success = CompileVerificationResult.success
    let failed = CompileVerificationResult.failed(errors: [])

    #expect(success.isSuccess == true)
    #expect(failed.isSuccess == false)
  }

  @Test("CompileError stores all properties")
  func compileErrorProperties() {
    let error = CompileError(
      line: 10,
      column: 5,
      message: "cannot find type 'Foo'",
      severity: .error
    )

    #expect(error.line == 10)
    #expect(error.column == 5)
    #expect(error.message.contains("Foo"))
    #expect(error.severity == .error)
  }

  @Test("CompileVerificationError descriptions are helpful")
  func errorDescriptions() {
    let notFound = CompileVerificationError.compilerNotFound
    #expect(notFound.description.contains("Swift compiler"))

    let timeout = CompileVerificationError.timeout
    #expect(timeout.description.contains("timeout"))

    let errors = [CompileError(line: 5, column: 1, message: "test error", severity: .error)]
    let failed = CompileVerificationError.compilationFailed(errors: errors)
    #expect(failed.description.contains("Line 5"))
    #expect(failed.description.contains("test error"))
  }
}
