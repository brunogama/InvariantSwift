```swift
/// Protocol for intercepting generator and shrinking operations.
///
/// Implement this protocol to add cross-cutting concerns like logging,
/// validation, or metrics collection to generators.
public protocol GeneratorInterceptor: Sendable {
  /// Called when a value is generated.
  ///
  /// - Parameters:
  ///   - value: The generated value
  ///   - size: The size parameter used for generation
  /// - Returns: The value (possibly modified for validation interceptors)
  func onGenerate<T: Sendable>(_ value: T, size: Size) -> T

  /// Called when a value is shrunk.
  ///
  /// - Parameters:
  ///   - original: The original value being shrunk
  ///   - shrunk: The shrunk candidate
  ///   - step: The shrink step number
  /// - Returns: The shrunk value
  func onShrink<T: Sendable>(_ original: T, shrunk: T, step: Int) -> T

  /// Called when a property is evaluated.
  ///
  /// - Parameters:
  ///   - value: The input value
  ///   - passed: Whether the property passed
  func onPropertyEvaluated<T: Sendable>(_ value: T, passed: Bool)
}

// Default implementations (no-op)
extension GeneratorInterceptor {
  public func onGenerate<T: Sendable>(_ value: T, size: Size) -> T { value }
  public func onShrink<T: Sendable>(_ original: T, shrunk: T, step: Int) -> T { shrunk }
  public func onPropertyEvaluated<T: Sendable>(_ value: T, passed: Bool) {}
}
```

```swift
/// Interceptor that logs all generated and shrunk values.
public final class LoggingInterceptor: GeneratorInterceptor, @unchecked Sendable {
  private let output: @Sendable (String) -> Void
  private let includeSize: Bool
  private let includeShrinkSteps: Bool

  public init(
    output: @escaping @Sendable (String) -> Void = { print($0) },
    includeSize: Bool = true,
    includeShrinkSteps: Bool = true
  ) { ... }

  public func onGenerate<T: Sendable>(_ value: T, size: Size) -> T {
    let msg = includeSize
      ? "Generated: \(value) (size: \(size.value))"
      : "Generated: \(value)"
    output(msg)
    return value
  }

  public func onShrink<T: Sendable>(_ original: T, shrunk: T, step: Int) -> T {
    if includeShrinkSteps {
      output("Shrink[\(step)]: \(original) -> \(shrunk)")
    }
    return shrunk
  }

  public func onPropertyEvaluated<T: Sendable>(_ value: T, passed: Bool) {
    output("Evaluated: \(passed ? "PASS" : "FAIL") for \(value)")
  }
}
```

```swift
extension Gen {
  /// Attaches an interceptor to this generator.
  ///
  /// The interceptor's hooks are called during generation and shrinking.
  /// Multiple interceptors can be chained by calling this method multiple times.
  ///
  /// - Parameter interceptor: The interceptor to attach
  /// - Returns: A new generator with the interceptor attached
  public func withInterceptor(_ interceptor: some GeneratorInterceptor) -> Gen<T> {
    Gen<T>(
      generate: { rng, size in
        let value = self.generate(&rng, size)
        return interceptor.onGenerate(value, size: size)
      },
      shrink: Shrink<T> { value in
        self.shrink.shrink(value).enumerated().map { idx, shrunk in
          interceptor.onShrink(value, shrunk: shrunk, step: idx)
        }
      }
    )
  }
}
```

2. Multiple interceptors (chain):
```swift
extension Gen {
  /// Attaches multiple interceptors in order.
  ///
  /// Interceptors are called in the order provided.
  public func withInterceptors(_ interceptors: [any GeneratorInterceptor]) -> Gen<T> {
    interceptors.reduce(self) { gen, interceptor in
      gen.withInterceptor(interceptor)
    }
  }
}
```

3. Convenience method for logging:
```swift
extension Gen {
  /// Attaches a logging interceptor for debugging.
  ///
  /// - Parameter output: Custom output function (default: print)
  public func logged(output: @escaping @Sendable (String) -> Void = { print($0) }) -> Gen<T> {
    withInterceptor(LoggingInterceptor(output: output))
  }
}
```

```swift
extension Gen {

    let metrics = MetricsInterceptor()
    return (withInterceptor(metrics), metrics)
  }
}
```
