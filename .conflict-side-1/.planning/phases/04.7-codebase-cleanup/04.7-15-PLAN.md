   ```swift
   import Foundation

   /// Protocol for types that can be pretty-printed
   public protocol PrettyPrintable {
     /// Convert to pretty-printed string
     func prettyPrint() -> String

     /// Convert to pretty-printed string with indentation
     func prettyPrint(indent: Int) -> String
   }

   extension PrettyPrintable {
     /// Default implementation with no indentation
     public func prettyPrint() -> String {
       prettyPrint(indent: 0)
     }
   }

   // Conformances for common types
   extension String: PrettyPrintable {
     public func prettyPrint(indent: Int) -> String {
       String(repeating: "  ", count: indent) + "\"\(self)\""
     }
   }

   extension Int: PrettyPrintable {
     public func prettyPrint(indent: Int) -> String {
       String(repeating: "  ", count: indent) + "\(self)"
     }
   }

   extension Array: PrettyPrintable where Element: PrettyPrintable {
     public func prettyPrint(indent: Int) -> String {
       let prefix = String(repeating: "  ", count: indent)
       if isEmpty { return prefix + "[]" }

       var result = prefix + "[\n"
       for (index, element) in enumerated() {
         result += element.prettyPrint(indent: indent + 1)
         if index < count - 1 { result += "," }
         result += "\n"
       }
       result += prefix + "]"
       return result
     }
   }
   ```

   ```swift
   import Foundation

   /// Format for displaying diffs between values
   public enum DiffFormat: Sendable {
     case unified
     case sideBySide
     case inline
     case minimal
   }

   /// Result of a diff operation
   public struct DiffResult: Sendable {
     public let format: DiffFormat
     public let lines: [DiffLine]

     public init(format: DiffFormat, lines: [DiffLine]) {
       self.format = format
       self.lines = lines
     }

     public func render() -> String {
       switch format {
       case .unified:
         return renderUnified()
       case .sideBySide:
         return renderSideBySide()
       case .inline:
         return renderInline()
       case .minimal:
         return renderMinimal()
       }
     }

     private func renderUnified() -> String {
       lines.map { line in
         switch line {
         case .unchanged(let text): return "  \(text)"
         case .added(let text): return "+ \(text)"
         case .removed(let text): return "- \(text)"
         case .changed(let old, let new): return "- \(old)\n+ \(new)"
         }
       }.joined(separator: "\n")
     }

     private func renderSideBySide() -> String {
       lines.map { $0.description }.joined(separator: "\n")
     }

     private func renderInline() -> String {
       lines.map { $0.description }.joined(separator: " ")
     }

     private func renderMinimal() -> String {
       lines.compactMap { line in
         switch line {
         case .unchanged: return nil
         case .added(let text): return "+ \(text)"
         case .removed(let text): return "- \(text)"
         case .changed(let old, let new): return "~ \(old) → \(new)"
         }
       }.joined(separator: "\n")
     }
   }

   /// A single line in a diff
   public enum DiffLine: Sendable, CustomStringConvertible {
     case unchanged(String)
     case added(String)
     case removed(String)
     case changed(old: String, new: String)

     public var description: String {
       switch self {
       case .unchanged(let text): return "  \(text)"
       case .added(let text): return "+ \(text)"
       case .removed(let text): return "- \(text)"
       case .changed(let old, let new): return "~ \(old) → \(new)"
       }
     }
   }
   ```

   ```swift
   import Foundation

   /// Pretty printer for formatting test output
   public struct PrettyPrinter: Sendable {
     public let indent: Int
     public let maxWidth: Int

     public init(indent: Int = 0, maxWidth: Int = 100) {
       self.indent = indent
       self.maxWidth = maxWidth
     }

     /// Print a value with pretty formatting
     public func print<T: PrettyPrintable>(_ value: T) -> String {
       value.prettyPrint(indent: indent)
     }

     /// Print with custom indentation
     public func withIndent(_ level: Int) -> PrettyPrinter {
       PrettyPrinter(indent: level, maxWidth: maxWidth)
     }

     /// Format a diff between two values
     public func diff<T: Equatable & CustomStringConvertible>(
       _ lhs: T,
       _ rhs: T,
       format: DiffFormat = .unified
     ) -> DiffResult {
       let lhsStr = String(describing: lhs)
       let rhsStr = String(describing: rhs)

       let lhsLines = lhsStr.split(separator: "\n").map(String.init)
       let rhsLines = rhsStr.split(separator: "\n").map(String.init)

       var diffLines: [DiffLine] = []
       let maxLines = max(lhsLines.count, rhsLines.count)

       for i in 0..<maxLines {
         let lhsLine = i < lhsLines.count ? lhsLines[i] : nil
         let rhsLine = i < rhsLines.count ? rhsLines[i] : nil

         switch (lhsLine, rhsLine) {
         case (let lhs?, let rhs?) where lhs == rhs:
           diffLines.append(.unchanged(lhs))
         case (let lhs?, let rhs?):
           diffLines.append(.changed(old: lhs, new: rhs))
         case (let lhs?, nil):
           diffLines.append(.removed(lhs))
         case (nil, let rhs?):
           diffLines.append(.added(rhs))
         case (nil, nil):
           break
         }
       }

       return DiffResult(format: format, lines: diffLines)
     }
   }
   ```
