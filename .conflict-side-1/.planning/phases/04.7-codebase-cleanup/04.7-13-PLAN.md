2. Convert to actor:
   ```swift
   public actor ExampleDatabase {
     private var failingExamples: [TestIdentifier: [FailingExample]] = [:]
     private let storageURL: URL

     public init(storageURL: URL) {
       self.storageURL = storageURL
     }

     public func saveFailingExample(_ example: FailingExample, for testID: TestIdentifier) async throws {
       failingExamples[testID, default: []].append(example)
       try await persistToDisk()
     }

     public func loadFailingExamples(for testID: TestIdentifier) async throws -> [FailingExample] {
       if failingExamples.isEmpty {
         try await loadFromDisk()
       }
       return failingExamples[testID] ?? []
     }

     public func clear(for testID: TestIdentifier) async throws {
       failingExamples[testID] = nil
       try await persistToDisk()
     }

     private func persistToDisk() async throws {
       let data = try JSONEncoder().encode(failingExamples)
       try data.write(to: storageURL)
     }

     private func loadFromDisk() async throws {
       guard FileManager.default.fileExists(atPath: storageURL.path) else { return }
       let data = try Data(contentsOf: storageURL)
       failingExamples = try JSONDecoder().decode([TestIdentifier: [FailingExample]].self, from: data)
     }
   }
   ```

3. Verify compilation:

