@Suite("Value Collection Tests")
struct CollectTests {

  // MARK: - Basic Collection

  @Test("collect() tracks values in histogram")
  func collectTracksValues() async throws {
    let context = ClassificationContext()

    // Collect some values
    context.collect(5)
    context.collect(5)
    context.collect(10)
    context.collect(5)

    let report = context.report()

    // Verify histogram counts
    let collected = report.collectedValues["collected"]
    #expect(collected?["5"]?.count == 3)
    #expect(collected?["10"]?.count == 1)
  }

  @Test("collect() with custom category")
  func collectWithCategory() async throws {
    let context = ClassificationContext()

    context.collect("apple", category: "fruits")
    context.collect("banana", category: "fruits")
    context.collect("apple", category: "fruits")
    context.collect("carrot", category: "vegetables")

    let report = context.report()

    #expect(report.collectedValues["fruits"]?["apple"]?.count == 2)
    #expect(report.collectedValues["fruits"]?["banana"]?.count == 1)
    #expect(report.collectedValues["vegetables"]?["carrot"]?.count == 1)
  }

  @Test("collect() percentages calculated correctly")
  func collectPercentages() async throws {
    let context = ClassificationContext()

    for _ in 0..<75 { context.collect("a") }
    for _ in 0..<25 { context.collect("b") }

    let report = context.report()
    let collected = report.collectedValues["collected"]

    // 75 out of 100 = 75%
    #expect(collected?["a"]?.percentage ?? 0 >= 74.9)
    #expect(collected?["a"]?.percentage ?? 0 <= 75.1)

    context.tabulate("properties", labels: ["even", "small"])

    context.tabulate("properties", labels: ["even", "large"])

    let report = context.report()

    // Each label counted independently
    #expect(report.labelDistribution["properties"]?["even"]?.count == 2)
    #expect(report.labelDistribution["properties"]?["odd"]?.count == 1)
    #expect(report.labelDistribution["properties"]?["small"]?.count == 1)
    #expect(report.labelDistribution["properties"]?["large"]?.count == 2)

  func tabulateCategoriesIndependent() async throws {
    let context = ClassificationContext()

    context.tabulate("size", labels: ["small"])
    context.tabulate("color", labels: ["red"])
    context.tabulate("size", labels: ["large"])
    context.tabulate("color", labels: ["blue"])

    let report = context.report()

    // Size category
    #expect(report.labelDistribution["size"]?["small"]?.count == 1)
    #expect(report.labelDistribution["size"]?["large"]?.count == 1)

    #expect(report.labelDistribution["color"]?["red"]?.count == 1)
    #expect(report.labelDistribution["color"]?["blue"]?.count == 1)

@testable import InvariantSwift

@Suite("Counterexample Message Tests")
struct CounterexampleTests {

  // MARK: - Basic Message Storage

  @Test("addCounterexample() stores message closure")
  func addCounterexampleStoresClosure() async throws {
    let context = ClassificationContext()

    context.addCounterexample { (n: Int) in "Value was: \(n)" }

    let messages = context.computeCounterexampleMessages(42)
    #expect(messages.count == 1)
    #expect(messages[0] == "Value was: 42")
  }

  @Test("multiple counterexample closures accumulate")
  func counterexamplesAccumulate() async throws {
    let context = ClassificationContext()

    context.addCounterexample { (n: Int) in "First: \(n)" }
    context.addCounterexample { (n: Int) in "Second: \(n * 2)" }
    context.addCounterexample { (n: Int) in "Third: \(n * 3)" }

    let messages = context.computeCounterexampleMessages(10)
    #expect(messages.count == 3)
    #expect(messages[0] == "First: 10")
    #expect(messages[1] == "Second: 20")
    #expect(messages[2] == "Third: 30")
  }

  // MARK: - Lazy Evaluation

  @Test("counterexample closures not evaluated until needed")
  func counterexampleLazyEvaluation() async throws {
    var evaluationCount = 0
    let context = ClassificationContext()

    context.addCounterexample { (n: Int) in
      evaluationCount += 1
      return "Evaluated: \(n)"
    }

    // Closure stored but not yet evaluated
    #expect(evaluationCount == 0)

    // Now evaluate
    _ = context.computeCounterexampleMessages(42)
    #expect(evaluationCount == 1)

    // Calling again evaluates again (closures are not cached)
    _ = context.computeCounterexampleMessages(42)
    #expect(evaluationCount == 2)
  }

  // MARK: - Type Handling

  @Test("counterexample handles different types")
  func counterexampleDifferentTypes() async throws {
    let context = ClassificationContext()

    // Add closure for Int
    context.addCounterexample { (n: Int) in "Int: \(n)" }

    // Add closure for String
    context.addCounterexample { (s: String) in "String: \(s)" }

    // Int value - only Int closure matches
    let intMessages = context.computeCounterexampleMessages(42)
    #expect(intMessages.contains("Int: 42"))

    // String value - only String closure matches
    let stringMessages = context.computeCounterexampleMessages("hello")
    #expect(stringMessages.contains("String: hello"))
  }

  @Test("counterexample with struct type")
  func counterexampleWithStruct() async throws {
    struct User: Sendable {
      let name: String
      let age: Int
    }

    let context = ClassificationContext()

    context.addCounterexample { (user: User) in
      "User '\(user.name)' (age \(user.age)) failed validation"
    }

    let user = User(name: "Alice", age: -5)
    let messages = context.computeCounterexampleMessages(user)

    #expect(messages.count == 1)
    #expect(messages[0].contains("Alice"))
    #expect(messages[0].contains("-5"))
  }

  // MARK: - Empty Messages

  @Test("empty counterexample messages filtered out")
  func emptyMessagesFiltered() async throws {
    let context = ClassificationContext()

    context.addCounterexample { (n: Int) in n > 0 ? "Positive" : "" }
    context.addCounterexample { (n: Int) in "Always shown" }

    // Positive number - both messages
    let posMessages = context.computeCounterexampleMessages(5)
    #expect(posMessages.count == 2)

    // Non-positive - empty string filtered
    let negMessages = context.computeCounterexampleMessages(-5)
    #expect(negMessages.count == 1)
    #expect(negMessages[0] == "Always shown")
  }

  // MARK: - Thread Safety

  @Test("counterexample storage is thread-safe")
  func counterexampleThreadSafe() async throws {
    let context = ClassificationContext()
    let iterations = 100

    await withTaskGroup(of: Void.self) { group in
      for i in 0..<iterations {
        group.addTask {
          context.addCounterexample { (n: Int) in "Task \(i): \(n)" }
        }
      }
    }

    let messages = context.computeCounterexampleMessages(42)
    #expect(messages.count == iterations)
  }

  // MARK: - Integration

  @Test("counterexample integrates with classification")
  func counterexampleWithClassification() async throws {
    let context = ClassificationContext()

    // Add classification and counterexample
    context.classify("type", "integer")
    context.addCounterexample { (n: Int) in "Failed with: \(n)" }

    // Both should work independently
    let report = context.report()
    #expect(report.labelDistribution["type"]?["integer"]?.count == 1)

    let messages = context.computeCounterexampleMessages(42)
    #expect(messages.count == 1)
  }

  // MARK: - Formatting

  @Test("counterexample messages appear in failure format")
  func counterexampleInFailureFormat() async throws {
    let customMessages = [
      "Expected positive number",
      "Got: -5",
      "Suggestion: Use abs()"
    ]

    // Test the formatPropertyFailure function
    let result: PropertyResult<Int> = .failure(
      counterexample: -5,
      iterations: 10,
      shrunk: -5,
      reason: .predicateFailed,
      seed: Seed()
    )

    let formatted = formatPropertyFailure(
      result: result,
      customMessages: customMessages
    )

    #expect(formatted.contains("Expected positive number"))
    #expect(formatted.contains("Got: -5"))
    #expect(formatted.contains("Suggestion: Use abs()"))
    #expect(formatted.contains("Counterexample:"))
  }

  // MARK: - Dogfood Tests

  @Test("Dogfood: counterexample accumulation via property test")
  func dogfoodCounterexampleAccumulation() async throws {
    // Property: N addCounterexample calls should produce N messages
    let context = ClassificationContext()
    let count = 5

    for i in 0..<count {
      context.addCounterexample { (n: Int) in "Message \(i): \(n)" }
    }

    let messages = context.computeCounterexampleMessages(42)

    // Property holds: count of messages equals count of additions
    #expect(messages.count == count)

    // All messages should contain the value
    for message in messages {
      #expect(message.contains("42"))
    }
  }

  @Test("Dogfood: property test for message filtering")
  func dogfoodMessageFiltering() async throws {
    // Property: empty messages should always be filtered
    try await checkProperty(
      generator: Gen.int(in: -100...100),
      config: PropertyConfig(iterations: 50)
    ) { n in
      let context = ClassificationContext()
      context.addCounterexample { (v: Int) in v > 0 ? "positive" : "" }
      context.addCounterexample { (v: Int) in "always" }
      let messages = context.computeCounterexampleMessages(n)
      // Should have 2 messages if n > 0, 1 otherwise
      return n > 0 ? messages.count == 2 : messages.count == 1
    }
  }
