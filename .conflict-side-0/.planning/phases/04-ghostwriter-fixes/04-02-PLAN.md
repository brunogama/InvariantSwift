```swift
/// Result of attempting to generate a Gen expression for a type.
enum GeneratorResult {
  case success(String)  // Valid Gen expression
  case todoRequired(String, reason: String)  // Type name and why it can't be generated

  var expression: String {
    switch self {
    case .success(let expr): return expr
    case .todoRequired(let typeName, let reason):
      return "/* TODO: supply generator for \(typeName) - \(reason) */ fatalError()"
    }
  }
}
```

```swift
/// Generate the Gen expression for a type, returning TODO if not possible.
private func generatorResult(for typeName: String) -> GeneratorResult {
  var cleanedType = typeName
    .replacingOccurrences(of: "?", with: "")
    .replacingOccurrences(of: "!", with: "")
    .trimmingCharacters(in: .whitespaces)

  let isOptional = typeName.contains("?") || typeName.hasPrefix("Optional<")

  // Handle Optional<T>
  if cleanedType.hasPrefix("Optional<") && cleanedType.hasSuffix(">") {
    cleanedType = String(cleanedType.dropFirst(9).dropLast())
  }

  // Handle Array<T> or [T]
  if cleanedType.hasPrefix("Array<") && cleanedType.hasSuffix(">") {
    let inner = String(cleanedType.dropFirst(6).dropLast())
    let innerResult = generatorResult(for: inner)
    switch innerResult {
    case .success:
      return .success("composer.generate(using: Gen.array(of: \(inner).arbitrary))")
    case .todoRequired:
      return .todoRequired("[" + inner + "]", reason: "element type not generatable")
    }
  } else if cleanedType.hasPrefix("[") && cleanedType.hasSuffix("]") && !cleanedType.contains(":") {
    let inner = String(cleanedType.dropFirst().dropLast())
    let innerResult = generatorResult(for: inner)
    switch innerResult {
    case .success:
      return .success("composer.generate(using: Gen.array(of: \(inner).arbitrary))")
    case .todoRequired:
      return .todoRequired("[" + inner + "]", reason: "element type not generatable")
    }
  }
  // Handle Set<T>
  else if cleanedType.hasPrefix("Set<") && cleanedType.hasSuffix(">") {
    let inner = String(cleanedType.dropFirst(4).dropLast())
    let innerResult = generatorResult(for: inner)
    switch innerResult {
    case .success:
      return .success("Set(composer.generate(using: Gen.array(of: \(inner).arbitrary)))")
    case .todoRequired:
      return .todoRequired("Set<" + inner + ">", reason: "element type not generatable")
    }
  }
  // Handle Dictionary (limited support)
  else if cleanedType.hasPrefix("Dictionary<") || (cleanedType.hasPrefix("[") && cleanedType.contains(":")) {
    return .todoRequired(cleanedType, reason: "dictionary generation not yet supported")
  }
  // Standard types with known generators
  else if Self.knownGeneratableTypes.contains(cleanedType) {
    let baseExpr = "composer.generate(using: \(cleanedType).arbitrary)"
    if isOptional {
      return .success("composer.generate(using: Gen.optional(\(cleanedType).arbitrary))")
    }
    return .success(baseExpr)
  }
  // Unknown type
  else {
    return .todoRequired(cleanedType, reason: "unknown type - add Arbitrary conformance or use @Arbitrary attribute")
  }
}
```

Update `generateArbitraryExtension(for:)` to track whether TODOs are needed:

```swift
/// Result of generating Arbitrary extension.
public struct ArbitraryGenerationResult {
  let code: String
  let hasTodos: Bool
  let todoTypes: [String]
}

/// Generate Arbitrary extension with TODO tracking.
public func generateArbitraryExtensionResult(for type: ExtractedTypeInfo) -> ArbitraryGenerationResult {
  var todoTypes: [String] = []
  var propGenerators: [String] = []

  for prop in type.properties {
    let result = generatorResult(for: prop.typeName)
    switch result {
    case .success(let expr):
      propGenerators.append("\(prop.name): \(expr)")
    case .todoRequired(let typeName, _):
      todoTypes.append(typeName)
      propGenerators.append("\(prop.name): \(result.expression)")
    }
  }

  let code = """
    extension \(type.name): Arbitrary {
      public static var arbitrary: Gen<\(type.name)> {
        Gen.compose { composer in
          \(type.name)(
            \(propGenerators.joined(separator: ",\n        "))
          )
        }
      }
    }
    """

  return ArbitraryGenerationResult(
    code: code,
    hasTodos: !todoTypes.isEmpty,
    todoTypes: todoTypes
  )
}

/// Backward compatible method.
public func generateArbitraryExtension(for type: ExtractedTypeInfo) -> String {
  generateArbitraryExtensionResult(for: type).code
}
```

```swift
/// Check if a type can have Arbitrary auto-generated.
/// Returns true even if some properties need TODOs (will be marked in output).
static func canAutoGenerateArbitrary(for type: ExtractedTypeInfo) -> Bool {
  // Must have at least one property
  guard !type.properties.isEmpty else { return false }

  // At least one property must be generatable for it to be worth generating
  return type.properties.contains { prop in
    isPropertyTypeGeneratable(prop.typeName)
  }
}

/// Check if all properties are fully generatable (no TODOs needed).
static func canFullyGenerateArbitrary(for type: ExtractedTypeInfo) -> Bool {
  guard !type.properties.isEmpty else { return false }
  return type.properties.allSatisfy { prop in
    isPropertyTypeGeneratable(prop.typeName)
  }
}
```
