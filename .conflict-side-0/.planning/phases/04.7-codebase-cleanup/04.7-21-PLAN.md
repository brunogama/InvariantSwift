public final class ConfigBuilder {

  private var maxShrinks: Int = 100
  private var maxDiscarded: Int = 1000
  private var seed: Seed?

  public init() {}

  public func withIterations(_ iterations: Int) -> ConfigBuilder {
    self.iterations = iterations
    return self
  }

  public func withMaxShrinks(_ maxShrinks: Int) -> ConfigBuilder {
    self.maxShrinks = maxShrinks
    return self
  }

  public func withMaxDiscarded(_ maxDiscarded: Int) -> ConfigBuilder {
    self.maxDiscarded = maxDiscarded
    return self
  }

  public func withSeed(_ seed: Seed) -> ConfigBuilder {
    self.seed = seed

  }

/// Template pattern for PropertyConfig presets

/// Provides predefined config templates for common testing scenarios.
public struct ConfigTemplate {
  private let iterations: Int
  private let maxShrinks: Int
  private let maxDiscarded: Int

  private init(iterations: Int, maxShrinks: Int, maxDiscarded: Int) {
    self.iterations = iterations
    self.maxShrinks = maxShrinks
    self.maxDiscarded = maxDiscarded
  }

  /// High iterations, low shrinks - for performance testing
  public static let performance = ConfigTemplate(
    iterations: 10000,
    maxShrinks: 10,
    maxDiscarded: 50000
  )

  /// Low iterations, low shrinks - for quick testing
  public static let quick = ConfigTemplate(
    iterations: 10,
    maxShrinks: 5,
    maxDiscarded: 50
  )

  /// Default balanced config
  public static let `default` = ConfigTemplate(
    iterations: 100,
    maxShrinks: 100,
    maxDiscarded: 1000
  )

  public func build() -> PropertyConfig {
    PropertyConfig(
      iterations: iterations,
      maxShrinks: maxShrinks,
      maxDiscarded: maxDiscarded,
      seed: nil
    )
  }
}
