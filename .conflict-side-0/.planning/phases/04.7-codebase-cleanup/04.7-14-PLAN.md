   ```swift
   import Foundation

   extension PropertyConfig {
     /// Quick testing configuration (fewer iterations for fast feedback)
     public static func quickConfig(iterations: Int = 10) -> PropertyConfig {
       var config = PropertyConfig.default
       return PropertyConfig.iterations.set(config, iterations)
     }

     /// Performance testing configuration (many iterations with timeout)
     public static func performanceConfig(
       iterations: Int = 10000,
       timeout: TimeInterval = 60.0
     ) -> PropertyConfig {
       var config = PropertyConfig.default
       config = PropertyConfig.iterations.set(config, iterations)

       // Set timeout if PropertyConfig has timeout field
       if let timeoutLens = PropertyConfig.timeout {
         config = timeoutLens.set(config, timeout)
       }

       return config
     }

     /// Stress testing configuration (maximum iterations and shrinks)
     public static func stressConfig(
       iterations: Int = 100000,
       maxShrinks: Int = 10000
     ) -> PropertyConfig {
       var config = PropertyConfig.default
       config = PropertyConfig.iterations.set(config, iterations)
       config = PropertyConfig.maxShrinks.set(config, maxShrinks)
       return config
     }

     /// Development configuration (verbose output, moderate iterations)
     public static func devConfig(iterations: Int = 100) -> PropertyConfig {
       var config = PropertyConfig.default
       config = PropertyConfig.iterations.set(config, iterations)
       config = PropertyConfig.verbose.set(config, true)
       return config
     }
   }
   ```
