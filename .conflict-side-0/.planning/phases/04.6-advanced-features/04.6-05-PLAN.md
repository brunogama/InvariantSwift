extension ShrinkTree {
  /// Finds the minimal value satisfying a predicate using parallel exploration.
  ///
  /// Uses Swift Concurrency to explore multiple branches of the shrink tree
  /// concurrently. When multiple minimal candidates are found, selects
  /// deterministically to ensure reproducible results.
  ///
  /// - Parameters:
  ///   - budget: Maximum total nodes to visit across all workers
  ///   - workers: Number of concurrent workers (default: processor count)
  ///   - satisfying: Predicate that must be satisfied
  ///
  /// - Returns: The minimal satisfying value, or nil if none found
  @available(macOS 10.15, iOS 13.0, tvOS 13.0, watchOS 6.0, *)
  public func findMinimalParallel(
    budget: Int,
    workers: Int = ProcessInfo.processInfo.processorCount,
    satisfying predicate: @escaping @Sendable (T) async -> Bool
  ) async -> T? {
    // Verify root satisfies predicate
    guard await predicate(value) else { return nil }

    // For small trees, use sequential (overhead not worth it)
    if budget < 100 || workers <= 1 {
      return await findMinimalAsync(budget: budget, satisfying: predicate)
    }

    return await withTaskGroup(of: T?.self) { group in
      var best: T = value
      let budgetPerWorker = budget / workers
      let childBatches = children.chunked(into: workers)

      // Spawn workers for each batch of children
      for batch in childBatches {
        group.addTask {
          var localBest: T?
          for child in batch {
            if let found = await child.findMinimalAsync(
              budget: budgetPerWorker,
              satisfying: predicate
            ) {
              if localBest == nil {
                localBest = found
              } else {
                // Keep the "smaller" one (deterministic selection)
                localBest = self.selectSmaller(localBest!, found)
              }
            }
          }
          return localBest
        }
      }

      // Collect results
      for await result in group {
        if let found = result {
          best = selectSmaller(best, found)
        }
      }

      return best
    }
  }

  /// Deterministic selection between two minimal candidates.
  private func selectSmaller(_ a: T, _ b: T) -> T {
    // Default: prefer first found (stable sort property)
    // Subclasses can override for type-specific comparison
    return a
  }
}

// Helper for chunking arrays
extension Array {
  func chunked(into count: Int) -> [[Element]] {
    guard count > 0 else { return [self] }
    let chunkSize = (self.count + count - 1) / count
    return stride(from: 0, to: self.count, by: chunkSize).map {
      Array(self[$0..<Swift.min($0 + chunkSize, self.count)])
    }
  }
}

/// Actor for coordinating parallel shrink exploration.
///
/// Manages worker count, budget allocation, and result aggregation
/// for parallel shrinking operations.
@available(macOS 10.15, iOS 13.0, tvOS 13.0, watchOS 6.0, *)
public actor ParallelShrinker {

  // MARK: - Configuration

  public struct Config: Sendable {
    /// Number of concurrent workers
    public let workers: Int

    /// Total shrink budget
    public let budget: Int

    /// Minimum tree size to use parallel (below this, use sequential)
    public let parallelThreshold: Int

    /// Enable progress tracking
    public let trackProgress: Bool

    public init(
      workers: Int = ProcessInfo.processInfo.processorCount,
      budget: Int = 1000,
      parallelThreshold: Int = 100,
      trackProgress: Bool = false
    ) { ... }

    public static let `default` = Config()
  }

  // MARK: - State

  private let config: Config
  private var nodesVisited: Int = 0
  private var candidatesFound: Int = 0

  // MARK: - Initialization

  public init(config: Config = .default) {
    self.config = config
  }

  // MARK: - Public Interface

  /// Finds the minimal value in a shrink tree satisfying a predicate.
  public func findMinimal<T: Sendable>(
    in tree: ShrinkTree<T>,
    satisfying predicate: @escaping @Sendable (T) async -> Bool
  ) async -> T? {
    nodesVisited = 0
    candidatesFound = 0

    if config.budget < config.parallelThreshold {
      // Use sequential for small budgets
      return await tree.findMinimalAsync(
        budget: config.budget,
        satisfying: predicate
      )
    }

    return await tree.findMinimalParallel(
      budget: config.budget,
      workers: config.workers,
      satisfying: predicate
    )
  }

  /// Returns progress metrics.
  public func getProgress() -> (nodesVisited: Int, candidatesFound: Int) {
    (nodesVisited, candidatesFound)
  }
}
```

extension ParallelShrinker {
  /// Compares parallel vs sequential shrinking performance.
  ///
  /// Returns timing and result comparison for debugging/optimization.
  public func benchmark<T: Sendable & Equatable>(
    tree: ShrinkTree<T>,
    predicate: @escaping @Sendable (T) async -> Bool
  ) async -> ShrinkBenchmark<T> {
    let sequentialStart = CFAbsoluteTimeGetCurrent()
    let sequentialResult = await tree.findMinimalAsync(
      budget: config.budget,
      satisfying: predicate
    )
    let sequentialTime = CFAbsoluteTimeGetCurrent() - sequentialStart

    let parallelStart = CFAbsoluteTimeGetCurrent()
    let parallelResult = await tree.findMinimalParallel(
      budget: config.budget,
      workers: config.workers,
      satisfying: predicate
    )
    let parallelTime = CFAbsoluteTimeGetCurrent() - parallelStart

    return ShrinkBenchmark(
      sequentialResult: sequentialResult,
      sequentialTime: sequentialTime,
      parallelResult: parallelResult,
      parallelTime: parallelTime,
      speedup: sequentialTime / parallelTime
    )
  }
}

public struct ShrinkBenchmark<T: Sendable>: Sendable {
  public let sequentialResult: T?
  public let sequentialTime: TimeInterval
  public let parallelResult: T?
  public let parallelTime: TimeInterval
  public let speedup: Double
}
```
